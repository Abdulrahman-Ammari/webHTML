import subprocess, smtplib, re

def Send_Email(email, password, massegemail):
    server = smtplib.SMTP('smtp.gmail.com', 587)
    server.starttls()
    server.login(email, password)
    # Send as UTF-8 encoded bytes to avoid UnicodeEncodeError
    server.sendmail(email, email, massegemail.encode('utf-8'))
    server.quit()
